#include <stdio.h>
#include <string.h>
#include <ctype.h>
#define MAX 100

/* ================== STRUCT ================== */

struct Student {
    int id;
    char name[100];//maximum number of characters for the name is 100
    int age;
    float subject1;
    float subject2;
    float subject3;
    float average;
};
void capitalizeName(char name[])
{
    for(int i =0; name[i] != '\0'; i++){
        if(i == 0|| name[i -1] == ' '){
            name[i]= toupper((unsigned char)name[i]);
        }
    }
}
/* ================== ALL VARIABLES ================== */
struct Student students[MAX];
int count = 0;

/* ================== ADD STUDENT ================== */
void addStudent() {
    if (count >= MAX) {
        printf("Student list is full!\n");
        return;
    }

    printf("\nEnter Student ID: ");
    scanf("%d", &students[count].id);

    printf("Enter Student Name: ");
    getchar();
    fgets(students[count].name, sizeof(students[count].name), stdin);
    students[count].name[strcspn(students[count].name, "\n")]= '\0';
    capitalizeName(students[count].name);

    printf("Enter Age: ");
    scanf("%d", &students[count].age);

    printf("Enter Marks for subject 1: ");
    scanf("%f", &students[count].subject1);

    printf("Enter Marksforsubject 2: ");
    scanf("%f", &students[count].subject2);

    printf("Enter Marksforsubject 3: ");
    scanf("%f", &students[count].subject3);

    students[count].average = (students[count].subject1 + students[count].subject2 + students[count].subject3) / 3;

    count++;
    printf("Student added successfully!\n");
}

/* ================== DISPLAY STUDENTS ================== */
void displayStudents() {

    if (count == 0) {
        printf("\nNo students found.\n");
        return;
    }

    printf("\n===== STUDENT LIST =====\n");

    printf("ID\tName\t\tAge\tSub1\tSub2\tSub3\tAverage\n");
    printf("-------------------------------------------------------------\n");

    for (int i = 0; i < count; i++) {

        printf("%d\t%s\t%d\t%.2f\t%.2f\t%.2f\t%.2f\n" , students[i].id, students[i].name,  students[i].age, students[i].subject1, students[i].subject2, students[i].subject3,  students[i].average
        );
    }
}

/* ================== SEARCH STUDENT ================== */
void searchStudent() {
    int id;
    printf("\nEnter ID to search: ");
    scanf("%d", &id);

    for (int i = 0; i < count; i++) {
        if (students[i].id == id) {
            printf("%d\t%s\t%d\t%.2f\t%.2f\t%.2f\t%.2f\n", students[i].id, students[i].name, students[i].age, students[i].subject1, students[i].subject2, students[i].subject3, students[i].average);
            return;
        }
    }

    printf("Student not found.\n");
}

/* ================== DELETE STUDENT ================== */
void deleteStudent() {
    int id;
    printf("\nEnter ID to delete: ");
    scanf("%d", &id);

    for (int i = 0; i < count; i++) {
        if (students[i].id == id) {

            for (int j = i; j < count - 1; j++) {
                students[j] = students[j + 1];
            }

            count--;
            printf("Student deleted successfully!\n");
            return;
        }
    }

    printf("Student not found.\n");
}
/*----------menu for displaying the best student----------*/

{
    if (count == 0)

        return ;

    int best=0;
